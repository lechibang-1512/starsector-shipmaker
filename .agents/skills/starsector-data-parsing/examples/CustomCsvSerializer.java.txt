package shipeditor.parsing.saving;

import com.fasterxml.jackson.core.JsonGenerator;
import com.fasterxml.jackson.databind.JsonSerializer;
import com.fasterxml.jackson.databind.SerializerProvider;
import java.io.IOException;
import java.util.Map;

/**
 * Custom serializer preventing standard Jackson quotes on comment columns (#).
 */
public class CustomCsvSerializer extends JsonSerializer<Map<?, ?>> {

    @Override
    public void serialize(Map<?, ?> value, JsonGenerator gen, SerializerProvider serializers)
            throws IOException {
        
        gen.writeStartObject();
        for (Map.Entry<?, ?> entry : value.entrySet()) {
            Object key = entry.getKey();
            Object val = entry.getValue();
            
            if (val != null) {
                gen.writeStringField(key.toString(), val.toString());
            } else {
                gen.writeStringField(key.toString(), "");
            }
        }
        gen.writeEndObject();
    }
}
